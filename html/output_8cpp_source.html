9void writeToOutput(char *const dest, const std::string &src, const size_t outputSize,
10 logger::LogFlags logFlag) {
11 if (!dest) {
12 return;
13 }
14 if (src.empty())
15 return;
16 LOG_DEBUG("outputSize is {}, src.size() is {}", outputSize, src.size());
17 assert(src.size() <= outputSize);
18#ifdef _WIN32
19 // if count is _TRUNCATE, then as much of src as will it into dest is copied
20 // while still leaving room for the terminating null, which is always
21 // appended.
22 assert(outputSize > 0);
23 strncpy_s(dest, outputSize, src.c_str(), _TRUNCATE);
24#elif __linux__
25 // this function will take the full size of the buffer(not just the length)
26 // and guarantee to NUL-terminate the result (as long as size is larger than
27 // 0)
28 strlcpy(dest, src.c_str(), outputSize);
29#endif
30 LOG(logFlag, "writeToOutput: {}", src);
31}
