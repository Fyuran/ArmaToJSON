10void writeToOutput(char *const dest, const std::string &src, const size_t bufferSize,
11 ArmaToJSON::LogFlags logFlag) {
12 if (!dest) {
13 return;
14 }
15 if (src.empty())
16 return;
17 LOG_DEBUG("src is {} bytes long", src.size());
18 LOG_DEBUG("bufferSize is {}", bufferSize);
19 assert(src.size() <= bufferSize);
20#ifdef _WIN32
21 LOG(logFlag, "OUTPUT WIN32:{}", src);
22 // if count is _TRUNCATE, then as much of src as will it into dest is copied
23 // while still leaving room for the terminating null, which is always
24 // appended.
25 assert(bufferSize > 0);
26 strncpy_s(dest, bufferSize, src.c_str() + 1, _TRUNCATE);
27#elif __linux__
28 LOG(logFlag, "OUTPUT LINUX:{}", src);
29 // this function will take the full size of the buffer(not just the length)
30 // and guarantee to NUL-terminate the result (as long as size is larger than
31 // 0)
32 strlcpy(dest, src.c_str(), src.size() + 1);
33#endif
34}
